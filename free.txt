-- path แบบเดียวกับที่ใช้เขียนสคริปต์ได้จริง (แก้บัค escape เดิมแล้ว)
local function getPath(inst)
    if inst == nil then return "nil" end
    if inst == game then return "game" end

    local chain, cur, guard = {}, inst, 0
    while cur and cur ~= game and guard < 200 do
        guard = guard + 1
        table.insert(chain, 1, cur)
        cur = cur.Parent
    end

    local out
    if cur == game then out = "game" else out = "<detached>" end

    for i, obj in ipairs(chain) do
        local name = obj.Name
        local piece
        if i == 1 and cur == game and obj == workspace then
            out = "workspace"
        elseif i == 1 and cur == game and isService(obj) then
            piece = ':GetService("' .. obj.ClassName .. '")'
        elseif obj == LocalPlayer then
            piece = ".LocalPlayer"
        elseif name:match("^[%a_][%w_]*$") and not KEYWORDS[name] then
            piece = "." .. name
        else
            piece = "[" .. string.format("%q", name) .. "]"
        end
        if piece then out = out .. piece end
    end
    return out
end

local function rootOf(inst)
    local cur, guard = inst, 0
    while cur and cur.Parent and cur.Parent ~= game and guard < 200 do
        guard = guard + 1
        cur = cur.Parent
    end
    return (cur and cur.Name) or "?"
end

local function r3(n) return string.format("%.3f", n) end

local function fmt(v)
    local t = typeof(v)
    if t == "nil" then
        return "nil"
    elseif t == "string" then
        local s, extra = v, ""
        if #s > CONFIG.MAX_VALUE_LEN then
            extra = " ...(+" .. (#s - CONFIG.MAX_VALUE_LEN) .. " chars)"
            s = s:sub(1, CONFIG.MAX_VALUE_LEN)
        end
        s = s:gsub("%c", " ")
        return '"' .. s .. '"' .. extra
    elseif t == "Instance" then
        return getPath(v)
    elseif t == "Vector3" then
        return "Vector3.new(" .. r3(v.X) .. ", " .. r3(v.Y) .. ", " .. r3(v.Z) .. ")"
    elseif t == "Vector2" then
        return "Vector2.new(" .. r3(v.X) .. ", " .. r3(v.Y) .. ")"
    elseif t == "CFrame" then
        local c = { v:GetComponents() }
        for i = 1, #c do c[i] = r3(c[i]) end
        return "CFrame.new(" .. table.concat(c, ", ") .. ")"
    elseif t == "Color3" then
        return "Color3.fromRGB(" .. math.floor(v.R * 255 + 0.5) .. ", "
            .. math.floor(v.G * 255 + 0.5) .. ", " .. math.floor(v.B * 255 + 0.5) .. ")"
    elseif t == "table" then
        local ok, enc = pcall(HttpService.JSONEncode, HttpService, v)
        local s = ok and enc or tostring(v)
        if #s > CONFIG.MAX_VALUE_LEN then s = s:sub(1, CONFIG.MAX_VALUE_LEN) .. "..." end
        return s
    else
        return tostring(v)
    end
end

local function get(inst, prop)
    local ok, v = pcall(function() return inst[prop] end)
    if ok then return v end
    return nil
end

-- OUTPUT (เขียนไฟล์เป็นก้อนๆ กันกินแรม)
------------------------------------------------------------------
local Out = { buf = {}, path = nil, lines = 0, bytes = 0, canAppend = (appendfile ~= nil), tee = false, sbuf = {} }

local function flush(force)
    if not Out.path or not Out.canAppend then return end
    if #Out.buf == 0 then return end
    if force or #Out.buf >= 1500 then
        local chunk = table.concat(Out.buf, "\n") .. "\n"
        local ok = pcall(appendfile, Out.path, chunk)
        if ok then
            Out.buf = {}
        else
            Out.canAppend = false -- เก็บไว้ในแรม แล้วเขียนรวดเดียวตอนจบ
        end
    end
end

local function w(s)
    s = tostring(s)
    Out.buf[#Out.buf + 1] = s
    if Out.tee then Out.sbuf[#Out.sbuf + 1] = s end
    Out.lines = Out.lines + 1
    Out.bytes = Out.bytes + #s + 1
    if Out.path and Out.canAppend and #Out.buf >= 1500 then flush(false) end
end

local ops = 0
local function yieldMaybe()
    ops = ops + 1
    if ops % CONFIG.YIELD_EVERY == 0 then task.wait() end
end

local function section(title)
    w("")
    w(string.rep("=", 72))
    w(" " .. title)
    w(string.rep("=", 72))
end

local function kv(key, value, indent)
    indent = indent or ""
    local s
    if type(value) == "string" then
        s = value:gsub("[\r\n]+", " ")
    else
        s = fmt(value)
    end
    w(string.format("%s%-24s: %s", indent, key, s))
end

local function props(inst, list, indent)
    for _, p in ipairs(list) do
        local v = get(inst, p)
        if v ~= nil then kv(p, v, indent) end
    end
end

local function attrs(inst, indent)
    local ok, a = pcall(inst.GetAttributes, inst)
    if ok and type(a) == "table" then
        local keys = {}
        for k in pairs(a) do keys[#keys + 1] = k end
        table.sort(keys)
        for _, k in ipairs(keys) do kv("@" .. k .. " <" .. typeof(a[k]) .. ">", a[k], indent) end
    end
end

local function wrapJoin(items, indent, width)
    local line = indent
    for i, it in ipairs(items) do
        local piece = it .. (i < #items and ", " or "")
        if #line + #piece > width and #line > #indent then
            w((line:gsub("%s+$", "")))
            line = indent
        end
        line = line .. piece
    end
    if #line > #indent then w(line) end
end

local function joinPath(dir, name)
    if dir:sub(-1) ~= "/" then dir = dir .. "/" end
    return dir .. name
end

-- ลองเซฟที่ SAVE_DIR ก่อน ถ้าไม่ได้ค่อยเซฟใน workspace ของ executor
local function chooseOutputPath(fileName)
    if not writefile then return nil end
    local candidates = {}
    if CONFIG.SAVE_DIR and CONFIG.SAVE_DIR ~= "" then
        candidates[#candidates + 1] = joinPath(CONFIG.SAVE_DIR, fileName)
    end
    candidates[#candidates + 1] = fileName

    for _, p in ipairs(candidates) do
        local ok = pcall(writefile, p, "-- Remote & Map Dump\n")
        if ok then
            local exists = true
            if isfile then
                local ok2, r = pcall(isfile, p)
                exists = ok2 and r
            end
            if exists then return p end
        end
    end
    return nil
end

local myPos = getRootPos()

local function posOf(inst)
    local cur, n = inst, 0
    while cur and n < 30 do
        n = n + 1
        if cur:IsA("BasePart") then return cur.Position end
        if cur:IsA("Attachment") then return cur.WorldPosition end
        if cur:IsA("Model") then
            local ok, cf = pcall(cur.GetPivot, cur)
            if ok then return cf.Position end
        end
        cur = cur.Parent
    end
    return nil
end

local function whereIs(inst)
    local ok, p = pcall(posOf, inst)
    if ok and p then
        local s = fmt(p)
        if myPos then s = s .. "   (ห่างเรา " .. string.format("%.1f", (p - myPos).Magnitude) .. " studs)" end
        return s
    end
    return nil
end

local function describeListeners(remote)
    if not (CONFIG.SCAN_LISTENERS and getconnections) then return nil end
    local cn = remote.ClassName
    local sig
    if cn == "RemoteEvent" or cn == "UnreliableRemoteEvent" then
        sig = remote.OnClientEvent
    elseif cn == "BindableEvent" then
        sig = remote.Event
    end
    if not sig then return nil end

    local ok, conns = pcall(getconnections, sig)
    if not ok or type(conns) ~= "table" then return nil end

    local res, owners = {}, {}
    for i, c in ipairs(conns) do
        local okf, fn = pcall(function() return c.Function end)
        local desc = "(ไม่ทราบ function)"
        if okf and type(fn) == "function" then
            local oki, src, line, name, nparams, isvar = pcall(debug.info, fn, "slna")
            if oki then
                desc = "src=" .. tostring(src) .. "  line=" .. tostring(line) .. "  name=" .. tostring(name ~= "" and name or "?")
                    .. "  params=" .. tostring(nparams) .. (isvar and "+..." or "")
            end
            local okE, owner = pcall(function()
                local env = getfenv(fn)
                return env and env.script
            end)
            if okE and typeof(owner) == "Instance" then
                local op = getPath(owner)
                desc = desc .. "  script=" .. op
                owners[#owners + 1] = op
            end
        end
        local en = get(c, "Enabled")
        res[#res + 1] = "#" .. i .. " enabled=" .. tostring(en) .. "  " .. desc
    end
    return res, owners
end

local REMOTE_CLASSES = { RemoteEvent = true, RemoteFunction = true, UnreliableRemoteEvent = true }
local BINDABLE_CLASSES = { BindableEvent = true, BindableFunction = true }
local VALUE_CLASSES  = { StringValue = true, IntValue = true, NumberValue = true, BoolValue = true, ObjectValue = true, Vector3Value = true, CFrameValue = true, Color3Value = true, BrickColorValue = true }
local SCRIPT_CLASSES = { Script = true, LocalScript = true, ModuleScript = true }

local B = {
    remotes = {}, bindables = {}, values = {}, prompts = {}, clicks = {}, touch = {}, scripts = {},
    tools = {}, buttons = {}, textboxes = {}, labels = {}, seats = {}, spawns = {}, anims = {}, attrs = {},
}
local classCount, serviceCount, totalDesc = {}, {}, 0

local playerGui = LocalPlayer:FindFirstChildOfClass("PlayerGui") or LocalPlayer:WaitForChild("PlayerGui", 5)

local function scanService(svc)
    local ok, desc = pcall(svc.GetDescendants, svc)
    if not ok then return end
    serviceCount[svc.Name] = #desc

    for i = 1, #desc do
        local v = desc[i]
        local cn = v.ClassName
        totalDesc = totalDesc + 1
        classCount[cn] = (classCount[cn] or 0) + 1

        if REMOTE_CLASSES[cn] then
            B.remotes[#B.remotes + 1] = v
        elseif BINDABLE_CLASSES[cn] then
            -- Bindable ใน UI/Particle ของ Players เยอะมากและไม่มีประโยชน์ เลยข้าม
            if svc.Name ~= "Players" then B.bindables[#B.bindables + 1] = v end
        elseif VALUE_CLASSES[cn] then
            B.values[#B.values + 1] = v
        elseif cn == "ProximityPrompt" then
            B.prompts[#B.prompts + 1] = v
        elseif cn == "ClickDetector" then
            B.clicks[#B.clicks + 1] = v
        elseif cn == "TouchTransmitter" then
            B.touch[#B.touch + 1] = v
        elseif SCRIPT_CLASSES[cn] then
            B.scripts[#B.scripts + 1] = v
        elseif cn == "Tool" or cn == "HopperBin" then
            B.tools[#B.tools + 1] = v
        elseif (cn == "TextButton" or cn == "ImageButton") and playerGui and v:IsDescendantOf(playerGui) then
            B.buttons[#B.buttons + 1] = v
        elseif cn == "TextBox" and playerGui and svc.Name == "Players" and v:IsDescendantOf(playerGui) then
            B.textboxes[#B.textboxes + 1] = v
        elseif cn == "TextLabel" and playerGui and svc.Name == "Players" and v:IsDescendantOf(playerGui) then
            B.labels[#B.labels + 1] = v
        elseif cn == "Seat" or cn == "VehicleSeat" then
            B.seats[#B.seats + 1] = v
        elseif cn == "SpawnLocation" then
            B.spawns[#B.spawns + 1] = v
        elseif cn == "Animation" then
            B.anims[#B.anims + 1] = v
        end

        if CONFIG.SCAN_ATTRIBUTES then
            local okA, a = pcall(v.GetAttributes, v)
            if okA and type(a) == "table" and next(a) ~= nil then
                B.attrs[#B.attrs + 1] = v
            end
        end

        yieldMaybe()
    end
end

        w("(ไม่พบ)")
        return
    end
    local shown = #list
    if CONFIG.MAX_LIST > 0 and shown > CONFIG.MAX_LIST then shown = CONFIG.MAX_LIST end
    for i = 1, shown do
        local inst = list[i]
        w("")

        if not ok then w(I .. "(อ่านค่าไม่ได้: " .. tostring(err) .. ")") end

        w("... เหลืออีก " .. (#list - shown) .. " รายการ ไม่ได้แสดง (เพิ่ม CONFIG.MAX_LIST)")
    end
end

local function callCode(inst)
    local p = getPath(inst)
    local cn = inst.ClassName
    if cn == "RemoteEvent" or cn == "UnreliableRemoteEvent" then
        return p .. ":FireServer()"
    elseif cn == "RemoteFunction" then
        return p .. ":InvokeServer()"
    elseif cn == "BindableEvent" then
        return p .. ":Fire()   -- local เท่านั้น"
    else
        return p .. ":Invoke()   -- local เท่านั้น"
    end
end

local function writeRemotes(list)
    section("6. REMOTES / BINDABLES  (ทั้งหมด: " .. #list .. ")")

    local byClass, byRoot = {}, {}
    for _, r in ipairs(list) do
        byClass[r.ClassName] = (byClass[r.ClassName] or 0) + 1
        local rn = rootOf(r)
        byRoot[rn] = (byRoot[rn] or 0) + 1
    end
    w("-- แยกตามชนิด --")
    for k, v in pairs(byClass) do w(I .. k .. ": " .. v) end
    w("-- แยกตามที่อยู่ --")
    for k, v in pairs(byRoot) do w(I .. k .. ": " .. v) end

    local handlers, hOrder = {}, {}
    for i, inst in ipairs(list) do
        w("")
        w("[" .. i .. "] " .. inst.ClassName)
        kv("Name", inst.Name, I)
        kv("Path", getPath(inst), I)
        local par = inst.Parent
        kv("Parent", par and (par.Name .. " (" .. par.ClassName .. ")") or "nil", I)
        kv("Root service", rootOf(inst), I)
        kv("Call", callCode(inst), I)
        local ls, owners = describeListeners(inst)
        local hint
        if inst.ClassName == "RemoteFunction" then
            hint = "client -> server (InvokeServer รอค่าตอบกลับ)"
        elseif ls and #ls > 0 then
            hint = "server -> client (มี listener ฝั่ง client " .. #ls .. " ตัว)"
        elseif ls then
            hint = "น่าจะ client -> server (ไม่มี listener ฝั่ง client)"
        else
            hint = "ไม่ทราบ (executor ไม่มี getconnections)"
        end
        kv("Direction hint", hint, I)
        if ls then
            kv("Client listeners", #ls, I)
            for _, d in ipairs(ls) do w(I .. I .. d) end
            for _, o in ipairs(owners or {}) do
                if not handlers[o] then
                    handlers[o] = {}
                    hOrder[#hOrder + 1] = o
                end
                table.insert(handlers[o], inst.Name)
            end
        end
        attrs(inst, I)
        yieldMaybe()
    end

    if #hOrder > 0 then
        w("")
        w("-- LISTENER MAP: สคริปต์ฝั่ง client ที่รับ remote (บอกว่า remote ไหนเกี่ยวกับระบบไหน) --")
        table.sort(hOrder)
        for _, o in ipairs(hOrder) do
            w(o .. "  [" .. #handlers[o] .. "]")
            wrapJoin(handlers[o], I, 150)
        end
    end

    -- ลิสต์ก๊อปวางแบบเดิม (เฉพาะ remote ที่ยิงไปเซิร์ฟเวอร์)
    w("")
    w("-- COPY-PASTE LIST (เฉพาะ RemoteEvent / RemoteFunction) --")
    local n = 0
    for _, inst in ipairs(list) do
        local cn = inst.ClassName
        if cn == "RemoteEvent" or cn == "UnreliableRemoteEvent" or cn == "RemoteFunction" then
            n = n + 1
            w(n .. ".")
            w(callCode(inst))
            w("")
        end
    end
end

local function writeAttributes(list)
    section("8. ATTRIBUTES  (instances ที่มี attribute: " .. #list .. ")")
    if #list == 0 then w("(ไม่พบ)") return end
    local shown = #list
    if CONFIG.MAX_LIST > 0 and shown > CONFIG.MAX_LIST then shown = CONFIG.MAX_LIST end
    for i = 1, shown do
        w("")
        w("[" .. i .. "] " .. getPath(list[i]) .. "  (" .. list[i].ClassName .. ")")
        attrs(list[i], I)
        yieldMaybe()
    end
    if shown < #list then w("... เหลืออีก " .. (#list - shown) .. " รายการ") end
end

local function treeNode(inst, depth)
    local indent = string.rep("  ", depth)
    local label = "[" .. inst.ClassName .. "] " .. inst.Name
    if VALUE_CLASSES[inst.ClassName] then label = label .. " = " .. fmt(get(inst, "Value")) end
    w(indent .. label)
    yieldMaybe()

    local kids = inst:GetChildren()
    if depth >= CONFIG.TREE_DEPTH then
        if #kids > 0 then w(indent .. "  ... (" .. #kids .. " children ถูกซ่อนด้วย TREE_DEPTH)") end
        return
    end
    local shown = math.min(#kids, CONFIG.TREE_MAX_CHILDREN)
    for i = 1, shown do treeNode(kids[i], depth + 1) end
    if #kids > shown then w(indent .. "  ... +" .. (#kids - shown) .. " children อื่นๆ") end
end

local function writeTree()
    section("11. INSTANCE TREE  (depth " .. CONFIG.TREE_DEPTH .. ")")
    for _, svc in ipairs(game:GetChildren()) do
        if not CONFIG.SKIP_SERVICES[svc.Name] then
            local ok, err = pcall(treeNode, svc, 0)
            if not ok then w("(tree error " .. svc.Name .. ": " .. tostring(err) .. ")") end
        end
    end
end

local function uiPath(inst)
    local parts, cur, n = {}, inst, 0
    while cur and n < 40 do
        n = n + 1
        table.insert(parts, 1, cur.Name)

        cur = cur.Parent
    end
    return table.concat(parts, " > ")

-- ข้อความของ label/ปุ่มที่อยู่ใกล้ๆ ช่องนั้น ใช้เดาว่าช่องนี้คืออะไร
local function nearbyTexts(inst)
    local found, seen = {}, {}
    local cur, depth = inst.Parent, 0
    while cur and depth < 3 do
        depth = depth + 1
        local okK, kids = pcall(cur.GetChildren, cur)
        if okK then
            for _, k in ipairs(kids) do
                if k ~= inst and (k.ClassName == "TextLabel" or k.ClassName == "TextButton") then
                    local t = get(k, "Text")
                    if type(t) == "string" and t ~= "" and #t <= 60 and not seen[t] then
                        seen[t] = true
                        found[#found + 1] = k.Name .. "=" .. fmt(t)
                        if #found >= 8 then return found end
                    end
                end
            end
        end
        cur = cur.Parent
        if cur and cur:IsA("LayerCollector") then break end
    end
    return found
end

local function fetchWebInfo()
    if not CONFIG.FETCH_WEB_INFO then return nil end
    local req = request or http_request or (syn and syn.request) or (http and http.request)
    if not req then return nil end
    local okR, res = pcall(req, { Url = "https://games.roblox.com/v1/games?universeIds=" .. tostring(game.GameId), Method = "GET" })
    if not okR or type(res) ~= "table" or type(res.Body) ~= "string" then return nil end
    local okJ, data = pcall(HttpService.JSONDecode, HttpService, res.Body)
    if okJ and type(data) == "table" and type(data.data) == "table" and type(data.data[1]) == "table" then
        return data.data[1]
    end
    return nil
end

local parentCache = {}
local function parentKey(inst)
    local p = inst.Parent
    if not p then return "nil" end
    local k = parentCache[p]
    if not k then
        k = getPath(p)
        parentCache[p] = k
    end
    return k
end

local function groupByParent(list)
    local groups, order = {}, {}
    for _, inst in ipairs(list) do
        local key = parentKey(inst)
        local g = groups[key]
        if not g then
            g = {}
            groups[key] = g
            order[#order + 1] = key
        end
        g[#g + 1] = inst
        yieldMaybe()
    end
    table.sort(order)
    return groups, order
end

    w("Position ............. พิกัดโลก (studs)   dist = ระยะห่างจากตัวเรา")
    w("UI path .............. ลำดับชั้นใน ScreenGui เช่น Main > Shop > CodeBox")

    w("9i TEXT BOXES ........ ช่องกรอกข้อความทั้งหมด (Text = ค่าที่กรอกอยู่ / PlaceholderText = ข้อความจาง ๆ บอกว่าให้กรอกอะไร)")

local savedPath, finalStatus, summaryName, gameName

-- รันแต่ละหมวดแยกกัน: หมวดไหนพัง หมวดอื่นยังทำงานต่อ
local function run(name, fn, ...)
    local okR, errR = pcall(fn, ...)
    if not okR then
        w("")
        w("!!! หมวด [" .. name .. "] ผิดพลาด: " .. tostring(errR))
        warn("[Dumper] section " .. name .. " failed: " .. tostring(errR))
    end
end

local function main()
    myPos = getRootPos()
    setStatus("1/4 Preparing file...")

    -- 1) เตรียมไฟล์ก่อนสแกน (จะได้รู้ทันทีว่าเซฟได้ไหม)
    local info
    local okI, res = pcall(MarketplaceService.GetProductInfo, MarketplaceService, game.PlaceId)
    if okI and type(res) == "table" then info = res end
    if not info then
        local okI2, res2 = pcall(MarketplaceService.GetProductInfo, MarketplaceService, game.PlaceId, Enum.InfoType.Asset)
        if okI2 and type(res2) == "table" then info = res2 end
    end
    local web = fetchWebInfo()
    gameName = (info and info.Name) or (web and web.name) or nil

    local stamp = os.date("%Y%m%d_%H%M%S")
    local gname = tostring(gameName or get(game, "Name") or "Game")
    gname = gname:gsub("[^%w_%-]", "_"):sub(1, 40)
    local fileName = CONFIG.FILE_NAME or ("Dump_" .. gname .. "_" .. tostring(game.PlaceId) .. "_" .. stamp .. ".txt")
    summaryName = "Summary_" .. gname .. "_" .. tostring(game.PlaceId) .. "_" .. stamp .. ".txt"
    Out.path = chooseOutputPath(fileName)
    if Out.path then
        print("[Dumper] output file: " .. Out.path)
    else
        warn("[Dumper] writefile ใช้ไม่ได้ จะคัดลอกลง clipboard แทนตอนจบ")
    end

    Out.tee = true
    w(string.rep("#", 72))
    w("# Remote & Map Dumper v3")
    w("# Game : " .. tostring(gameName or "?") .. "  (PlaceId " .. tostring(game.PlaceId) .. ")")
    w("# Time : " .. os.date("%Y-%m-%d %H:%M:%S"))
    w(string.rep("#", 72))
    run("0 legend", writeLegend)

    -- 2) สแกนทั้งเกม
    setStatus("2/4 Scanning game... (may take a while)")
    for _, svc in ipairs(game:GetChildren()) do
        if not CONFIG.SKIP_SERVICES[svc.Name] then
            local okS, errS = pcall(scanService, svc)
            if not okS then warn("[Dumper] scan " .. svc.Name .. " failed: " .. tostring(errS)) end
        end
    end

    -- 3) เขียนข้อมูลทั้งหมด
    setStatus("3/4 Writing data... (" .. totalDesc .. " instances)")
    run("1 game info",      writeGameInfo, info, web)
    run("2 local player",   writeLocalPlayer)
    run("3 players/teams",  writePlayersAndTeams)
    run("4 world settings", writeWorldSettings)
    run("5 class summary",  writeClassSummary)
    run("6a remotes folder", writeRemotesByFolder, B.remotes)
    Out.tee = false
    run("6 remotes",        writeRemotes, B.remotes)
    run("6b bindables",     writeCompact, "6b. BINDABLES (นอก Players)  BE=BindableEvent  BF=BindableFunction", B.bindables, labelBindable)
    run("7 values",         writeValuesGrouped, B.values)
    run("8 attributes",     writeAttributesGrouped, B.attrs)
    run("9a prompts",       listSection, "9a. PROXIMITY PROMPTS", B.prompts, descPrompt)
    run("9b clickers",      listSection, "9b. CLICK DETECTORS", B.clicks, descClick)
    run("9c touch",         listSection, "9c. TOUCH INTERESTS (ชิ้นส่วนที่ตอบสนองเมื่อสัมผัส)", B.touch, descTouch)
    run("9d tools",         listSection, "9d. TOOLS", B.tools, descTool)
    run("9e buttons",       listSection, "9e. GUI BUTTONS (PlayerGui)", B.buttons, descButton)
    run("9i textboxes",     listSection, "9i. TEXT BOXES (ช่องกรอกข้อความ)", B.textboxes, descTextBox)
    run("9j labels",        writeLabels, B.labels)
    run("9f seats",         listSection, "9f. SEATS / VEHICLE SEATS", B.seats, descSeat)
    run("9g spawns",        listSection, "9g. SPAWN LOCATIONS", B.spawns, descSpawn)
    run("9h animations",    writeCompact, "9h. ANIMATIONS (Name=AnimationId)", B.anims, labelAnim)
    run("10 scripts",       writeCompact, "10. SCRIPTS  S=Script  L=LocalScript  M=ModuleScript  x=ปิดอยู่", B.scripts, labelScript)
    if CONFIG.DUMP_TREE then run("11 tree", writeTree) end

    Out.tee = true
    section("END")
    w("Remotes: " .. #B.remotes .. " | Bindables: " .. #B.bindables .. " | Values: " .. #B.values .. " | Prompts: " .. #B.prompts
        .. " | Clickers: " .. #B.clicks .. " | Scripts: " .. #B.scripts)
    w("Instances scanned: " .. totalDesc .. " | Time used: " .. string.format("%.1f", os.clock() - startClock) .. "s")
end

local ok, err = xpcall(main, function(e) return tostring(e) .. "\n" .. debug.traceback() end)
if not ok then
    w("")
    w("!!! ERROR ระหว่างดั้ม: " .. tostring(err))
    warn("[Dumper] error: " .. tostring(err))
end

-- 4) จบงาน: เขียนส่วนที่เหลือ / สำรองด้วย clipboard
setStatus("4/4 Saving...")
if Out.path and Out.canAppend then
    flush(true)
    savedPath = Out.path
elseif Out.path and not Out.canAppend and writefile then
    local okW = pcall(writefile, Out.path, table.concat(Out.buf, "\n") .. "\n")
    if okW then savedPath = Out.path end
end

-- ไฟล์สรุปสั้น
local summaryPath
if CONFIG.SUMMARY_FILE and #Out.sbuf > 0 and writefile and summaryName then
    local sp = chooseOutputPath(summaryName)
    if sp then
        local okSW = pcall(writefile, sp, table.concat(Out.sbuf, "\n") .. "\n")
        if okSW then summaryPath = sp end
    end
end

-- อ่านไฟล์กลับมาเช็คว่ามีจริงและมีข้อมูล
local verify = ""
if savedPath and readfile then
    local okR, content = pcall(readfile, savedPath)
    if okR and type(content) == "string" then
        verify = ", " .. #content .. " bytes"
    else
        verify = ", (cannot read back)"
    end
end

if savedPath then
    finalStatus = "DONE. Saved: " .. savedPath .. " (" .. Out.lines .. " lines" .. verify .. ")"
elseif setclipboard then
    pcall(setclipboard, table.concat(Out.buf, "\n"))
    finalStatus = "Could not save file -> copied to clipboard (" .. Out.lines .. " lines)"
else
    finalStatus = "FAILED: executor has no writefile and no setclipboard"
end
if summaryPath then finalStatus = finalStatus .. "  | Summary: " .. summaryPath end
if not ok then finalStatus = finalStatus .. "  [had errors, see file/console]" end

setStatus(finalStatus)
task.delay(20, function()
    pcall(function() statusLabel.Parent:Destroy() end)
end)
