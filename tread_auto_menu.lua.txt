local logLines = {}
local logHook = nil
local function log(text)
    table.insert(logLines, os.date("%H:%M:%S") .. "  " .. tostring(text))
    if #logLines > 60 then
        table.remove(logLines, 1)
    end
    if logHook then
        logHook(tostring(text))
    end
end

local function short(v)
    local s = tostring(v)
    if #s > 42 then
        s = string.sub(s, 1, 42) .. "…"
    end
    return s
end

local lastLog = {}
local function logOnce(key, text)
    if lastLog[key] ~= text then
        lastLog[key] = text
        log(text)
    end
end

-- รีโมตจากโมดูลเกม ถ้าไม่ได้ก็ไปหาตัวจริงใน Networking
local function remObj(ns, name)
    local o = R and R[ns] and R[ns][name]
    if o then
        return o
    end
    if net then
        for _, pre in ipairs({"RF", "RE", "URE"}) do
            local x = net:FindFirstChild(pre .. "/" .. ns .. "/" .. name)
            if x then
                return x
            end
        end
    end
    return nil
end

local function ask(ns, name, ...)
    local o = remObj(ns, name)
    if not o then
        return table.pack(false, "ไม่เจอรีโมต " .. ns .. "." .. name)
    end
    return table.pack(pcall(function(...)
        return o:InvokeServer(...)
    end, ...))
end

local function send(ns, name, ...)
    local o = remObj(ns, name)
    if not o then
        return false
    end
    local a = table.pack(...)
    return pcall(function()
        o:FireServer(table.unpack(a, 1, a.n))
    end)
end

local function listenRemote(ns, name, fn)
    local o = remObj(ns, name)
    if o and o.OnClientEvent then
        local ok, c = pcall(function()
            return o.OnClientEvent:Connect(fn)
        end)
        if ok then
            return c
        end
    end
    return nil
end

local SUF = {"", "K", "M", "B", "T", "Qa", "Qi", "Sx", "Sp", "Oc", "No", "Dc"}
local function fmt(n)
    n = tonumber(n) or 0
    local i = 1
    while n >= 1000 and i < #SUF do
        n = n / 1000
        i = i + 1
    end
    if i == 1 then
        return string.format("%d", math.floor(n))
    end
    return string.format("%.2f%s", n, SUF[i])
end

-- ล็อกการเคลื่อนที่ ไม่ให้ฟังก์ชันอื่นดึงตัวกลับไปลู่วิ่งระหว่างทำงาน
local moveLock = 0
local function withMove(fn)
    moveLock = moveLock + 1
    local ok, err = pcall(fn)
    moveLock = moveLock - 1
    if not ok then
        log("พลาด: " .. short(err))
    end
end

local function waitUntil(cond, timeout)
    local t0 = os.clock()
    while os.clock() - t0 < (timeout or 3) do
        local ok, v = pcall(cond)
        if ok and v then
            return true
        end
        task.wait(0.1)
    end
    return false
end

-- ============ กดปุ่มใน UI เกม ============
local function press(btn)
    if not btn then
        return false
    end
    local did = false
    for _, sigName in ipairs({"Activated", "MouseButton1Click", "MouseButton1Down", "MouseButton1Up"}) do
        local sig = btn[sigName]
        if getconnections then
            pcall(function()
                for _, c in ipairs(getconnections(sig)) do
                    pcall(function()
                        c:Fire()
                    end)
                    did = true
                end
            end)
        elseif firesignal then
            if pcall(firesignal, sig) then
                did = true
            end
        end
    end
    return did
end

local function walk(root, ...)
    local cur = root
    for _, n in ipairs({...}) do
        if not cur then
            return nil
        end
        cur = cur:FindFirstChild(n)
    end
    return cur
end

local function setBox(box, text)
    if not box then
        return
    end
    box.Text = tostring(text)
    pcall(function()
        if firesignal then
            firesignal(box.FocusLost, true)
        end
    end)
end

local function firePrompt(p)
    if not p then
        return false
    end
    if fireproximityprompt then
        local ok = pcall(fireproximityprompt, p)
        if ok then
            return true
        end
    end
    return false
end

local rarityRank, rarityNames = {}, {}
if AssetsData and AssetsData.Directory then
    for _, d in pairs(AssetsData.Directory) do
        local r = d.Rarity
        if r then
            local n = r.DisplayName or r._id
            if n and rarityRank[n] == nil then
                rarityRank[n] = r.Rank or 0
                table.insert(rarityNames, n)
            end
        end
    end
    table.sort(rarityNames, function(a, b)
        return rarityRank[a] < rarityRank[b]
    end)
end
if #rarityNames == 0 then
    rarityNames = {"Common"}
    rarityRank.Common = 0
end
settings.sellRarity = rarityNames[1]

local function petInfo(enc)
    if not AssetItems then
        return nil
    end
    local ok, v = pcall(AssetItems.Decode, enc)
    if ok and type(v) == "table" then
        return v
    end
    return nil
end

local function rankOf(cat)
    local d = AssetsData and AssetsData.Directory and AssetsData.Directory[cat]
    local r = d and d.Rarity
    return (r and r.Rank) or 0, d
end

local function randomSpot(plot)
    local a = plot.PetArea
    local w = a.CFrame * CFrame.new((math.random() - 0.5) * a.Size.X * 0.8, a.Size.Y / 2, (math.random() - 0.5) * a.Size.Z * 0.8)
    return plot.CenterPoint.CFrame:ToObjectSpace(CFrame.new(w.Position))
end

end

local function countKeys(t)
    local n = 0

end

local plantBlock = 0
local function plantAll()
    if os.clock() < plantBlock or not EggState then
        return 0
    end
    local plot = myPlot()
    if not plot then
        return 0
    end
    local n = 0
    for uid, rec in pairs(ownedEggs()) do
        if rec.Placement == nil then
            local placed, why = false, nil
            for _ = 1, 6 do
                local ok, done, reason = pcall(EggState.PlantEgg, uid, randomSpot(plot))
                if ok and done then
                    placed = true
                    break
                end
                why = reason or done
                task.wait(0.15)
            end
            if placed then
                n = n + 1
            else
                plantBlock = os.clock() + 25
                logOnce("plant", "วางไข่ไม่ได้ " .. short(why))
                break
            end
        end
    end
    if n > 0 then
        log("วางไข่ลงฐาน " .. n .. " ฟอง")
    end
    return n
end

local function fieldPrompts(names)
    local out = {}
    for _, c in ipairs(workspace:GetChildren()) do
        if c.Name == "SmartPromptPart" and c:IsA("BasePart") then
            for _, p in ipairs(c:GetChildren()) do
                if p:IsA("ProximityPrompt") and names[p.Name] and p.Enabled then
                    table.insert(out, {part = c, prompt = p})
                end
            end
        end
    end
    return out
end

local function nearest(list)
    local h = hrp()
    if not h then
        return nil
    end
    local best, bd = nil, math.huge
    for _, it in ipairs(list) do
        local d = (it.part.Position - h.Position).Magnitude
        if d < bd then
            best, bd = it, d
        end
    end
    return best
end

local function bestAsset()
    if not (AssetsData and AssetsData.Directory) then
        return nil
    end
    local best, br, bw = nil, -1, 0
    for id, d in pairs(AssetsData.Directory) do
        local r = d.Rarity and d.Rarity.Rank or 0
        local w = d.DropWeight or 0
        if r > br or (r == br and w < bw) then
            best, br, bw = id, r, w
        end
    end
    return best
end

local function assetArg()
    local a = settings.asset
    if a == nil or a == "" then
        a = bestAsset()
    end
    return a
end

            flyBV:Destroy()

        end

            flyConn = RunService.RenderStepped:Connect(function()
                local h, hu = hrp(), hum()
                if not (h and hu) then
                    return
                end
                if not flyBV or flyBV.Parent ~= h then
                    if flyBV then
                        flyBV:Destroy()
                    end
                    flyBV = Instance.new("BodyVelocity")
                    flyBV.MaxForce = Vector3.new(1e6, 1e6, 1e6)
                    flyBV.Parent = h
                end
                local cam = workspace.CurrentCamera
                local dir = hu.MoveDirection
                local look = cam.CFrame.LookVector
                local move = Vector3.zero
                if dir.Magnitude > 0 then
                    local flat = Vector3.new(look.X, 0, look.Z)
                    local fwd = flat.Magnitude > 0 and dir:Dot(flat.Unit) or 0
                    move = dir + Vector3.new(0, look.Y * fwd, 0)
                end
                flyBV.Velocity = move * settings.flySpeed
            end)
        end
    end

end

        local function text()
            return tostring(settings[def.key]) .. (def.suffix or "")
        end
        local b = make("TextButton", {AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -14, 0.5, 0), Size = UDim2.fromOffset(92, 28), BackgroundColor3 = T.panel, BorderSizePixel = 0, Text = text(), TextSize = 12, TextColor3 = T.text, Font = Enum.Font.GothamMedium}, top)
        rounded(b, 8)
        outlined(b)
        b.Activated:Connect(function()
            local idx = 1
            for i, v in ipairs(def.options) do
                if v == settings[def.key] then
                    idx = i
                end
            end
            settings[def.key] = def.options[idx % #def.options + 1]
            b.Text = text()
        end)
    elseif def.kind == "input" then
        local box = make("TextBox", {AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -90, 0.5, 0), Size = UDim2.fromOffset(150, 28), BackgroundColor3 = T.panel, BorderSizePixel = 0, Text = "", PlaceholderText = def.hint or "", PlaceholderColor3 = T.muted, TextSize = 12, TextColor3 = T.text, Font = Enum.Font.Gotham, ClearTextOnFocus = false}, top)
        rounded(box, 8)
        outlined(box)
        box:GetPropertyChangedSignal("Text"):Connect(function()
            settings[def.key] = box.Text
        end)
        local b = make("TextButton", {AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -14, 0.5, 0), Size = UDim2.fromOffset(64, 28), BackgroundColor3 = T.orangeBg, BorderSizePixel = 0, Text = "ส่ง", TextSize = 12, TextColor3 = T.orange, Font = Enum.Font.GothamBold}, top)
        rounded(b, 8)
        outlined(b, T.orange)
        b.Activated:Connect(function()
            settings[def.key] = box.Text
            runFeature(def)
        end)
    end
    def.row = row
    return row
end

local function showTab(i)
    currentTab = i
    for idx, b in ipairs(tabButtons) do
        local active = idx == i
        b.BackgroundColor3 = active and T.panel2 or T.window
        b.TextColor3 = active and T.orange or T.muted
    end
    for _, def in ipairs(features) do
        def.row.Visible = def.tab == i
    end
    logRow.Visible = i == #TABS
    headLabel.Text = TABS[i]
    local n = 0
    for _, d in ipairs(rowsByTab[i] or {}) do
        if d.kind == "toggle" and d.on then
            n = n + 1
        end
    end
    headCount.Text = (i == #TABS) and (#logLines .. " รายการ") or (#(rowsByTab[i] or {}) .. " ฟังก์ชัน")
    scroller.CanvasPosition = Vector2.new(0, 0)
    if i == #TABS then
        logLabel.Text = #logLines > 0 and table.concat(logLines, "\n") or "ยังไม่มีอะไร"
    end
end

        end)
    end
end

local dragging, dragStart, startPos = false, nil, nil
titleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        if input.Position.X - titleBar.AbsolutePosition.X > 80 * scale.Scale and input.Position.X - titleBar.AbsolutePosition.X < titleBar.AbsoluteSize.X - 110 * scale.Scale then
            dragging = true
            dragStart = input.Position
            startPos = window.Position
        end
    end
end)
local moveConn = UIS.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local d = input.Position - dragStart
        window.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
    end
end)
local endConn = UIS.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)
